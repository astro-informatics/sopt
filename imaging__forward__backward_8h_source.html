331 template <typename SCALAR>
332 typename ImagingForwardBackward<SCALAR>::Diagnostic ImagingForwardBackward<SCALAR>::operator()(
333  t_Vector &out, t_Vector const &guess, t_Vector const &res) {
334  if(!g_function_)
335  {
336  throw std::runtime_error("Non-differentiable function `g` has not been set. You must set it with `g_function()` before calling the algorithm.");
337  }
338  g_function_->log_message();
339  Diagnostic result;
340  auto const g_proximal = g_function_->proximal_operator();
341  t_Gradient f_gradient;
342  Real gradient_step_size;
343  if(f_function_)
344  {
345  f_gradient = f_function_->gradient();
346  gradient_step_size = f_function_->get_step_size();
347  }
348  if(!f_gradient)
349  {
350  SOPT_MEDIUM_LOG("Gradient function has not been set; using default (gaussian likelihood) gradient. (To set a custom gradient set_gradient() must be called before the algorithm is run.)");
351  f_gradient = [this](t_Vector &output, t_Vector const &x, t_Vector const &residual, t_LinearTransform const &Phi) {
352  output = Phi.adjoint() * (residual / (this->sigma() * this->sigma()));
353  };
354  gradient_step_size = sigma()*sigma();
355  }
356  ScalarRelativeVariation<Scalar> scalvar(relative_variation(), relative_variation(),
357  "Objective function");
358  auto const convergence = [this, &scalvar](t_Vector const &x, t_Vector const &residual) mutable {
359  const bool result = this->is_converged(scalvar, x, residual);
360  this->objmin_ = std::real(scalvar.previous());
361  return result;
362  };
363  auto fb = ForwardBackward<SCALAR>(f_gradient, g_proximal, target())
364  .itermax(itermax())
365  .step_size(gradient_step_size)
366  .regulariser_strength(regulariser_strength())
367  .sq_op_norm(sq_op_norm())
368  .fista(fista())
369  .Phi(Phi())
370  .is_converged(convergence)
371  .random_updater(random_updater_)
372  .set_problem_state(problem_state);
373  static_cast<typename ForwardBackward<SCALAR>::Diagnostic &>(result) =
374  fb(out, std::tie(guess, res));
375  return result;
376 }

379 bool ImagingForwardBackward<SCALAR>::residual_convergence(t_Vector const &x,
380  t_Vector const &residual) const {
381  if (static_cast<bool>(residual_convergence())) return residual_convergence()(x, residual);
382  if (residual_tolerance() <= 0e0) return true;

388 template <typename SCALAR>
389 bool ImagingForwardBackward<SCALAR>::objective_convergence(ScalarRelativeVariation<Scalar> &scalvar,
390  t_Vector const &x,
391  t_Vector const &residual) const {
392  if (static_cast<bool>(objective_convergence())) return objective_convergence()(x, residual);
393  if (scalvar.relative_tolerance() <= 0e0) return true;
394  auto const current = ((regulariser_strength() > 0) ? g_function_->function(x)
395  * regulariser_strength() : 0) + \
396  ((f_function_) ? f_function_->function(x, target(), Phi()) : std::pow(sopt::l2_norm(residual), 2) / (2 * sigma() * sigma()));
397  return scalvar(current);
398 }

415 template <typename SCALAR>
416 bool ImagingForwardBackward<SCALAR>::is_converged(ScalarRelativeVariation<Scalar> &scalvar,
417  t_Vector const &x,
418  t_Vector const &residual) const {
419  auto const user = static_cast<bool>(is_converged()) == false or is_converged()(x, residual);
420  auto const res = residual_convergence(x, residual);
421 #ifdef SOPT_MPI
422  auto const obj = objective_convergence(obj_comm(), scalvar, x, residual);
423 #else
424  auto const obj = objective_convergence(scalvar, x, residual);
425 #endif

428  return user and res and obj;
